cmake_minimum_required(VERSION 3.24)
project(NetWatch VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "NetWatch requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
include_directories(src)
include_directories(SYSTEM vendor)
add_library(netwatch_core STATIC src/model.cpp src/config.cpp src/probes.cpp src/engine.cpp)
target_link_libraries(netwatch_core PUBLIC ws2_32 iphlpapi winhttp ole32)
add_library(NetWatch SHARED src/plugin.cpp src/settings.cpp src/resources.rc)
target_link_libraries(NetWatch PRIVATE netwatch_core comctl32 gdi32 user32)
enable_testing()
add_executable(netwatch_settings_tests tests/settings_tests.cpp src/resources.rc)
target_link_libraries(netwatch_settings_tests PRIVATE netwatch_core user32 comctl32)
add_test(NAME settings COMMAND netwatch_settings_tests)
set_tests_properties(settings PROPERTIES TIMEOUT 20)
add_executable(netwatch_tests tests/core_tests.cpp)
target_link_libraries(netwatch_tests PRIVATE netwatch_core)
add_test(NAME core COMMAND netwatch_tests)
add_executable(netwatch_host tests/host.cpp)
target_link_libraries(netwatch_host PRIVATE netwatch_core gdi32 user32 psapi)
set(NETWATCH_REFERENCE_HEADER "" CACHE FILEPATH "Independent official V1.86 header for the ABI host")
if(NETWATCH_REFERENCE_HEADER)
  get_filename_component(NETWATCH_REFERENCE_DIRECTORY "${NETWATCH_REFERENCE_HEADER}" DIRECTORY)
  target_include_directories(netwatch_host SYSTEM PRIVATE "${NETWATCH_REFERENCE_DIRECTORY}")
  target_compile_definitions(netwatch_host PRIVATE NW_HOST_INTERFACE="${NETWATCH_REFERENCE_HEADER}")
endif()
add_test(NAME plugin_host COMMAND netwatch_host $<TARGET_FILE:NetWatch> ${CMAKE_CURRENT_BINARY_DIR}/host-config)
add_test(NAME render_and_layout COMMAND netwatch_host $<TARGET_FILE:NetWatch> ${CMAKE_CURRENT_BINARY_DIR}/render-config --render)
set_tests_properties(render_and_layout PROPERTIES TIMEOUT 30)
set_tests_properties(core plugin_host PROPERTIES TIMEOUT 90)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME network COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/network.py $<TARGET_FILE:netwatch_tests>)
  add_test(NAME provenance COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/provenance.py)
  add_test(NAME http_unload COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/network.py $<TARGET_FILE:netwatch_host> $<TARGET_FILE:NetWatch> ${CMAKE_CURRENT_BINARY_DIR}/http-host-config)
  set_tests_properties(network PROPERTIES TIMEOUT 60)
  set_tests_properties(http_unload PROPERTIES TIMEOUT 60)
endif()
